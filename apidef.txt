def produce_message(key, value):
	# NOTE: Mode lokal - hasil crawling TIDAK dikirim ke Kafka/Database,
	# melainkan disimpan sebagai file JSON Lines di folder LOCAL_OUTPUT_DIR.
	try:
		file_name = os.path.join(
			LOCAL_OUTPUT_DIR,
			datetime.datetime.now().strftime("%Y-%m-%d") + ".jsonl"
		)
		with open(file_name, "a", encoding="utf-8") as f:
			f.write(json.dumps(value, default=str, ensure_ascii=False) + "\n")
		print(f"[INFO] Crawling result saved locally -> {file_name} (key={key})")
	except Exception as e:
		generate_log(
			4605,
			str(e),
			"Queue",
			{
				"client_id":os.environ["CLIENT_ID"],
				"error":str(e)
			}
		)

		4502:{"code":"Credentials Expired","level":"Error"},
		4503:{"code":"Failed to Store Data in Object Storage","level":"Error"},
		4504:{"code":"Account Blocked","level":"Error"},
		4601:{"code":"Database Connection Refused","level":"Critical"},
		4602:{"code":"No Route to Database Host","level":"Critical"},
		4603:{"code":"API Request Failed","level":"Critical"},
		4604:{"code":"Connection to Queue Failed","level":"Critical"},
		4605:{"code":"Queue Operation Failed","level":"Critical"},

	}

	log = {"code":None,"level":None}
	for key,value in logic_list.items():
		if key == status:
			log = value
			break

	duration = round((time.time()-start_time) * 1000)

def extract_update_urns_from_dom(driver, post_urls, seen):
	added = 0
	src = driver.page_source or ""

	ugc_patterns = [
		r'userGeneratedContentId=(\d{19})',
		r'urn:li:ugcPost:(\d{19})',
		r'userGeneratedContentPostUrn=UserGeneratedContentPostUrn\(userGeneratedContentId=(\d{19})\)',
	]

	share_patterns = [
		r'shareId=(\d{19})',
		r'urn:li:share:(\d{19})',
		r'ShareUrn\(shareId=(\d{19})\)',
	]

	ugc_ids = set()
	share_ids = set()

	for pattern in ugc_patterns:
		ugc_ids.update(re.findall(pattern, src))

	for pattern in share_patterns:
		share_ids.update(re.findall(pattern, src))

	for ugc_id in ugc_ids:
		raw_post_id = f"urn:li:ugcPost:{ugc_id}"
		url = f"https://www.linkedin.com/feed/update/{raw_post_id}/"

		if url not in seen:
			seen.add(url)
			post_urls.append(url)
			added += 1

	for share_id in share_ids:
		raw_post_id = f"urn:li:share:{share_id}"
		url = f"https://www.linkedin.com/feed/update/{raw_post_id}/"

		if url not in seen:
			seen.add(url)
			post_urls.append(url)
			added += 1

	return added

def scroll_search_results(driver):
	moved = False

	# 1) scroll main container
	try:
		workspace = driver.find_element(By.ID, "workspace")
		before = driver.execute_script("return arguments[0].scrollTop", workspace)
		driver.execute_script("arguments[0].scrollTop = arguments[0].scrollTop + 1200;", workspace)
		after = driver.execute_script("return arguments[0].scrollTop", workspace)

		# print(f"[DEBUG] workspace scrollTop before={before}, after={after}")
		moved = after > before
	except Exception as e:
		print(f"[DEBUG] workspace scroll failed: {e}")

	# 2) if there is load more button, click
	try:
		buttons = driver.find_elements(
			By.XPATH,
			"//button[contains(., 'Load more') or contains(., 'Muat lebih banyak')]"
		)

		for btn in buttons:
			try:
				if btn.is_displayed() and btn.is_enabled():
					driver.execute_script("arguments[0].scrollIntoView({block:'center'});", btn)
					driver.execute_script("arguments[0].click();", btn)
					# print("[DEBUG] clicked Load more")
					return True
			except Exception as e:
				print(f"[DEBUG] failed clicking one Load more button: {e}")
	except Exception as e:
		print(f"[DEBUG] Load more lookup failed: {e}")

	return moved

		
def crawling(keyword, client_id, scroll):
	check_login_status()
	post_urls = []
	max_pagination = 10
	page_count = 0
	seen = set()

	print(f"[INFO] Search Query: {urllib.parse.unquote(keyword)}")

	while True:
		page_count += 1
		search_url = (
			"https://www.linkedin.com/search/results/content/?keywords="
			+ keyword
			+ "&page="
			+ str(page_count)
			+ "&sortBy=\"date_posted\"&datePosted=\"past-24h\""
		)
		print(f"[INFO] Search URL: {search_url}")

		try:
			driver.get(search_url)
			time.sleep(5)

			# save_debug_screenshot(driver, f"page_{page_count}_after_get.png")

			added = extract_update_urns_from_dom(driver, post_urls, seen)
			print(f"[INFO] Added {added} urls (initial), total unique={len(post_urls)}")
			
			# save_debug_screenshot(driver, f"page_{page_count}_after_initial_extract.png")
			
			scroll_times = random.randint(5, 10)
			print(f"[INFO] Randomly decided to scroll {scroll_times} times.")

			# Scroll + extract per batch
			for i in range(scroll_times):
				try:
					moved = scroll_search_results(driver)
					time.sleep(random.uniform(4, 10))

					# save_debug_screenshot(driver, f"page_{page_count}_scroll_{i+1}.png")

					added = extract_update_urns_from_dom(driver, post_urls, seen)
					print(
						f"[INFO] Scroll {i+1}/{scroll_times}: "
						f"moved={moved}, added {added}, total unique={len(post_urls)}"
					)

				except Exception as e:
					print("[ERROR] Failed to scroll the page:", e)
					# save_debug_screenshot(driver, f"page_{page_count}_scroll_{i+1}_error.png")
					break

			print("[ERROR] Failed to crawl:", e)

					post_time_str = driver.find_element_by_xpath("//*[contains(@class, 'update-components-actor__sub-description') and contains(@class, 'text-body-xsmall')]").text.split(" •")[0].replace(" • Edited •   ","").replace(" • Diedit •   ","").replace("\n","").strip()
				except Exception as e:
					post_time_str = None
				print_blue(f"[DEBUG] Post date: {post_time_str}")


				try:
					if "mnt" in post_time_str:
						post_time_datetime = moment.date(post_time_str.replace("mnt", "minutes ago"))
					if "jam" in post_time_str:
						post_time_datetime = moment.date(post_time_str.replace("jam", "hours ago"))
					if "hr" in post_time_str:
						post_time_datetime = moment.date(post_time_str.replace("hr", "days ago"))
					if "mgg" in post_time_str:
						post_time_datetime = moment.date(post_time_str.replace("mgg", "weeks ago"))
					else:
						post_time_datetime = moment.date(post_time_str)
				except Exception as e:
					post_time_datetime = None
					print("[ERROR] Failed to get post_time_datetime:", e)

				try:
					# post_time_datetimems = int(post_time_datetime.strftime("%s"))*1000
					post_time_datetimems = int(post_time_datetime.datetime.timestamp() * 1000)
				except Exception as e:
					post_time_datetimems = None
			
			total_data += 1
			print(f"[DEBUG] [{total_data}] {post_id} | {post_time_str}")				
			
			# Metadata Crawling
			metadata = {
				"crawler": {
					"server_ip": server_ip,
					"git_commit_id": git_commit_id,
					"account": {
						"user": username,
						"token": None
					},
				"type": "login",
				"search": urllib.parse.unquote(keyword),
				"client_id": int(os.environ["CLIENT_ID"]),
				"platform": "Media Intelligence",
				"crawling_mode": mode,
				"author": "macan"
				}
			}
			
			# Insert Data LinkedIn Post
			insert_data = {
				"post_id": post_id,
				"url": url,
				"datetime_crawling_ms": datetime_crawling_ms,
				"owner": {
					"name": post_owner_name,
					"url": post_owner_url,
					"headline": post_owner_headline,
					"avatar": post_owner_pic
				},
				"post":{
					"content_str": content_str
				},
				"post_time":{
					"post_time_str": post_time_str,
					"post_time_datetime": str(post_time_datetime.date),
					"post_time_datetimems": post_time_datetimems
				},
				"datetime_ms": post_time_datetimems,
				"hashtag": hashtag,
				"comment_count": comment_count,
				"reaction_count": reaction_count,
				"metadata": metadata,
				"created_time": created_time,
				"updated_time": updated_time
			}

			# Sent data to Kafka
			produce_message(post_id, insert_data)
			
		except Exception as e:
			print("[ERROR] Reason:", e)			
		time.sleep(2)

	generate_log(
		0000,
		"Crawling finished for this loop. Please check the data to review total results.",
		"crawling summary",
		{
			"username": username,
			"client_id": client_id,
			"total_data": total_data,
			"keyword": keyword,
			"ip_server":server_ip
		}
	)

	return 1


def stop_user_username():
	try:
		accounts = _load_accounts()
		for acc in accounts:
			if acc.get("username") == username:
				acc["in_use"] = False
		_save_accounts(accounts)
	except Exception as e:
		print("[ERROR] Failed to update local account file.")
		generate_log(
			4601,
			"Failed to update local account file.",
			"login",
			{
				"client_id": os.environ["CLIENT_ID"],
				"error": str(e)
			}
		)

def kill_service(message):
	stop_user_username()
	username_failed()
	print(message)
	for prc in multiprocessing.active_children():
		prc.terminate()
	try:
		driver.close()
		driver.quit()
	except Exception as e:
		print(e)
	sys.exit(0)

HARDCODED_KEYWORDS = [
	"lowongan kerja data analyst",
	"startup teknologi indonesia",
	"digital marketing tips",
]

def get_target():
	print("[INFO] Start generating LinkedIn search targets.")

	datastore = []
	try:
		count_target = 0
		list_target = list(HARDCODED_KEYWORDS)

		random.shuffle(list_target)

		for query in list_target:
			count_target += 1
			keyword = urllib.parse.quote(query)
			client_id = str(os.environ["CLIENT_ID"])
			scroll = False

			print(f"[DEBUG] [{count_target}] Query search: {urllib.parse.unquote(keyword)}")
			
			datastore.append({
				"keyword": keyword,
				"scroll": scroll,
				"client_id": client_id
			})	
			
	except Exception as e:
		print("[ERROR] Failed to get Target:", e)
		generate_log(
			4501,
			"Unable to retrieve keywords",
			"login",
				{
					"client_id": os.environ["CLIENT_ID"],
					"error": str(e)
				}
			)

	if len(datastore) == 0:
			generate_log(
			4603,
			"Unable to retrieve keywords",
			"login",
				{
					"client_id": os.environ["CLIENT_ID"],
					"list_target_count": 0
				}
			)

	random.shuffle(datastore)
	print(f"[INFO] Finished generating {count_target} LinkedIn search targets.")

	return datastore
